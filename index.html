クラウド自動同期（Googleスプレッドシート連携）

未設定

Google Apps Script（GAS）のウェブアプリURLをPCとスマホの両方に設定すると、学習のたびに自動でクラウド保存・同期されます。

GAS設定用コードをコピー（初回のみ）

1. Googleスプレッドシートを新規作成 →「拡張機能」→「Apps Script」を開く
2. 下のコードを貼り付けて保存 →「デプロイ」→「新しいデプロイ」→ 種類「ウェブアプリ」、アクセスできるユーザー「全員」でデプロイし、発行されたURLを上に貼り付けます。 

var SECRET_KEY = ""; // 任意の合言葉を設定すると他人のアクセスを遮断できます

function checkKey(e) {
  if (!SECRET_KEY) return true;
  return e && e.parameter && e.parameter.key === SECRET_KEY;
}
function doGet(e) {
  if (!checkKey(e)) return ContentService.createTextOutput(JSON.stringify({error: "unauthorized"})).setMimeType(ContentService.MimeType.JSON);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  var vals = sheet.getRange("A1:A20").getValues().flat().filter(Boolean);
  var json = vals.join("") || "{}";
  return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}
function doPost(e) {
  if (!checkKey(e)) return ContentService.createTextOutput(JSON.stringify({error: "unauthorized"})).setMimeType(ContentService.MimeType.JSON);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  var data = e.postData.contents || "{}";
  sheet.getRange("A1:A20").clearContent();
  var chunkSize = 40000;
  for (var i = 0; i * chunkSize < data.length && i < 20; i++) {
    sheet.getRange(i + 1, 1).setValue(data.slice(i * chunkSize, (i + 1) * chunkSize));
  }
  return ContentService.createTextOutput(JSON.stringify({status: "ok"})).setMimeType(ContentService.MimeType.JSON);
}
